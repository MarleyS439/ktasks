cmake_minimum_required(VERSION 3.22)
project(ktasks VERSION 1.1.0 LANGUAGES CXX)

set(QT_MIN_VERSION "6.6.0")
set(KF_MIN_VERSION "6.5.0")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ECM ${KF_MIN_VERSION} REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})

include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(FeatureSummary)
include(ECMInstallIcons)

set(_ktasks_qt_components Core Gui Qml Quick QuickControls2 Sql Network)
set(_ktasks_kf_components CoreAddons Config I18n CalendarCore Notifications ColorScheme)
if(ANDROID)
    # Phones have no tray, printers dialogs or window system integration;
    # Kirigami on Android needs Qt Svg for its icons.
    list(APPEND _ktasks_qt_components Svg)
else()
    list(APPEND _ktasks_qt_components Widgets PrintSupport)
    list(APPEND _ktasks_kf_components StatusNotifierItem WindowSystem IconThemes)
endif()
if(UNIX AND NOT APPLE AND NOT ANDROID)
    list(APPEND _ktasks_qt_components DBus)
    list(APPEND _ktasks_kf_components DBusAddons GlobalAccel Crash)
endif()

find_package(Qt6 ${QT_MIN_VERSION} REQUIRED COMPONENTS ${_ktasks_qt_components})
find_package(OpenSSL REQUIRED)
find_package(KF6 ${KF_MIN_VERSION} REQUIRED COMPONENTS ${_ktasks_kf_components})
find_package(KF6Kirigami ${KF_MIN_VERSION} REQUIRED)
find_package(KF6KirigamiAddons 1.4 REQUIRED)
find_package(Qt6Keychain REQUIRED)
if(UNIX AND NOT APPLE AND NOT ANDROID)
    find_package(KPim6Akonadi QUIET COMPONENTS Core)
endif()

qt_policy(SET QTP0001 NEW)
if(QT_KNOWN_POLICY_QTP0004)
    qt_policy(SET QTP0004 NEW)
endif()

remove_definitions(-DQT_USE_QSTRINGBUILDER -DQT_NO_CAST_FROM_ASCII -DQT_NO_CAST_TO_ASCII -DQT_NO_CAST_FROM_BYTEARRAY -DQT_NO_URL_CAST_FROM_STRING)

add_subdirectory(src/core)
add_subdirectory(src/app)

if(ANDROID)
    option(BUILD_TESTING "Build the tests" OFF)
else()
    option(BUILD_TESTING "Build the tests" ON)
endif()
if(BUILD_TESTING)
    find_package(Qt6 ${QT_MIN_VERSION} REQUIRED COMPONENTS Test)
    enable_testing()
    add_subdirectory(tests)
endif()

ki18n_install(po)

if(ANDROID)
    install(FILES data/ktasks.notifyrc DESTINATION ${KDE_INSTALL_KNOTIFYRCDIR})
    install(FILES data/io.github.marleys439.ktasks.metainfo.xml DESTINATION ${KDE_INSTALL_METAINFODIR})
elseif(UNIX AND NOT APPLE)
    install(FILES data/io.github.marleys439.ktasks.desktop DESTINATION ${KDE_INSTALL_APPDIR})
    install(FILES data/io.github.marleys439.ktasks.metainfo.xml DESTINATION ${KDE_INSTALL_METAINFODIR})
    install(FILES data/ktasks.notifyrc DESTINATION ${KDE_INSTALL_KNOTIFYRCDIR})
    install(FILES data/io.github.marleys439.ktasks.svg DESTINATION ${KDE_INSTALL_ICONDIR}/hicolor/scalable/apps)
    install(FILES data/io.github.marleys439.ktasks-symbolic.svg DESTINATION ${KDE_INSTALL_ICONDIR}/hicolor/symbolic/apps)
    install(FILES data/io.github.marleys439.ktasks-symbolic.svg DESTINATION ${KDE_INSTALL_ICONDIR}/hicolor/scalable/status)
elseif(WIN32)
    install(FILES data/ktasks.notifyrc DESTINATION ${KDE_INSTALL_DATADIR}/knotifications6)
endif()

feature_summary(WHAT ALL INCLUDE_QUIET_PACKAGES FATAL_ON_MISSING_REQUIRED_PACKAGES)
